// Positive guarantees are checked as negative compiler tests.
function negativeChecks(received: ReceivedAccount) {
  // @ts-expect-error A successful Result must contain value.
  const missing: Result<number, string> = {ok:true};
  // @ts-expect-error The proposed phase requires a branded Candidate.
  const proposed: ProposedAccount = {...genesisAB(), _tag:'proposed'};
  // @ts-expect-error Proposing from open may not return received.
  const wrongPhase: Handler<AccountGrammar,'propose','open'> = () => ok(done(received));
  return [missing,proposed,wrongPhase];
}
