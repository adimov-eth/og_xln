 assert.equal(first.value.replica.accountReplicas.get(ALICE)?._tag,'open');
 assert.equal(first.value.replica.accountReplicas.get(ALICE)?.head.height,1n);

 report('F3 Entity altered retry',next);
}

 const voted=applyEntityInput({...e,state:imported},{kind:'txs',timestamp:NOW,txs:[{type:'vote',data:{proposalId:'id',voter:aliceAddr,choice:'yes'}}]},{...verifiers,self:ALICE,signerId:aliceAddr});
